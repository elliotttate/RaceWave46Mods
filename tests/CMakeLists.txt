cmake_minimum_required(VERSION 3.20)
project(MediaModTests LANGUAGES CXX)
enable_testing()
get_filename_component(ROOT ".." ABSOLUTE BASE_DIR "${CMAKE_CURRENT_LIST_DIR}")
if(NOT RECOMP_SDK_ROOT)
    message(FATAL_ERROR "Set RECOMP_SDK_ROOT to the RaceWave46 source checkout")
endif()
set(SDL_ROOT "${RECOMP_SDK_ROOT}/lib/rt64/src/contrib/mupen64plus-win32-deps/SDL2-2.26.3")
add_executable(media_mod_test media_mods/native_test.cpp "${ROOT}/mods/replacement_music/native.cpp")
target_compile_features(media_mod_test PRIVATE cxx_std_20)
target_compile_definitions(media_mod_test PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(media_mod_test PRIVATE "${ROOT}/include" "${RECOMP_SDK_ROOT}/lib/N64ModernRuntime/N64Recomp/include")
add_test(NAME native_music_mod_bridge COMMAND media_mod_test)
add_executable(music_test replacement_music/music_test.cpp "${ROOT}/src/wr64_music.cpp")
target_compile_features(music_test PRIVATE cxx_std_20)
target_compile_definitions(music_test PRIVATE NOMINMAX)
target_include_directories(music_test PRIVATE "${ROOT}/include" "${SDL_ROOT}/include" "${RECOMP_SDK_ROOT}/lib/rt64/src/contrib")
target_link_libraries(music_test PRIVATE "${SDL_ROOT}/lib/x64/SDL2.lib")
add_custom_command(TARGET music_test POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy_if_different
    "${SDL_ROOT}/lib/x64/SDL2.dll" "$<TARGET_FILE_DIR:music_test>")
foreach(scenario missing invalid mix bundled local-override invalid-local)
    add_test(NAME music_${scenario} COMMAND music_test ${scenario})
endforeach()
